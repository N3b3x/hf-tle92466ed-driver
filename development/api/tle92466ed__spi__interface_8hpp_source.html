266 [[nodiscard]] static constexpr SPIFrame MakeRead(uint16_t addr) noexcept {
267 SPIFrame frame{};

269 frame.tx_fields.address = 0; // Bits 23:17 are don't care for reads

891template <typename Derived>
892inline CommResult<uint32_t> SpiInterface<Derived>::Read(uint16_t address, bool verify_crc) noexcept {
893 // Create read frame
894 SPIFrame tx_frame = SPIFrame::MakeRead(address);
895
896 // Calculate and set CRC
897 tx_frame.tx_fields.crc = CalculateFrameCrc(tx_frame);
898
899 /* Pipelined SPI: the reply to a command only appears on the NEXT CS window,
900 * so the device always has one reply outstanding. That makes a bare
901 * command+dummy pair depend on what the previous transaction left behind —
902 * one dropped or extra frame anywhere shifts every later read by one slot,
903 * and the caller silently gets the *previous* register's value (bench
904 * 2026-08-12: PIN_STAT returning CH_CTRL's 0x8000, FB_VOLTAGE2 returning 0).
905 *
906 * Sending a leading dummy makes the transaction self-synchronising: it
907 * absorbs whatever reply was outstanding, so the reply to `tx_frame` is
908 * always in the last slot regardless of prior bus state.
909 *
910 * The dummies address 0, NOT `address`. Re-sending the same read as the
911 * dummy queues a second reply and the part answers with a reserved
912 * reply-mode instead of the register (bench 2026-08-12: ICVID never
913 * validated, DriverError::SPIFrameError every attempt; MakeRead(0) restored
914 * identity on the same hardware). TransferMulti keeps one bus-ownership
915 * window across all three frames on shared multi-slave buses. */
916 SPIFrame dummy_frame = SPIFrame::MakeRead(0);
917 dummy_frame.tx_fields.crc = CalculateFrameCrc(dummy_frame);
918
919 const uint32_t tx_words[3] = {dummy_frame.word, tx_frame.word,
920 dummy_frame.word};
921
922 /* A reply is only believed when it is well formed: CRC valid *and* the reply
923 * width the register is defined to answer with. Bench measurement
924 * (2026-08-13, 150 frames) found 4 % of frames corrupt and 25 % of
925 * FB_STAT reads carrying the preceding ICVID reply — the latter with a valid
926 * CRC, so CRC alone does not reject it. Both used to be accepted as data,
927 * which is the origin of the "sticky-zero CH_CTRL", the impossible IMAX, the
928 * open-load reported on a fitted resistor, and INIT_DONE reading 0 on a
929 * healthy device. Two CRC attempts keep freeze-complete (2× FB_UPD +
930 * FB_I_AVG + FB_DC) inside a 2 ms host control-loop budget; four attempts ×
931 * Mode1 IdleAll+30 µs per frame ran last_step ~9–11 ms on hold ticks.
932 * ICVID Init has its own recovery path. */
933 constexpr uint8_t kReadAttempts = 2;
934 const ExpectedReply expected = ExpectedReplyFor(address);
935 uint32_t rx_all[3] = {};
936 uint32_t last_reply = 0;
937 uint32_t last_flush = 0;
938 bool saw_any_frame = false;
939
940 for (uint8_t attempt = 0; attempt < kReadAttempts; ++attempt) {
941 auto multi = static_cast<Derived*>(this)->TransferMulti(
942 std::span<const uint32_t>{tx_words}, std::span<uint32_t>{rx_all});
943 if (!multi) {
944 return tle::unexpected(multi.error());
945 }
946 /* Slot 2 answers the command; slot 1 answers the flush dummy and is only
947 * kept as a fallback for the ICVID scan below. */
948 last_reply = rx_all[2];
949 last_flush = rx_all[1];
950
951 SPIFrame candidate{};
952 candidate.word = last_reply;
953
954 /* Undriven MISO — nothing to validate, and retrying is pointless if the
955 * part is absent, but cheap if a single window was missed. */
956 if (last_reply == 0U || last_reply == 0xFFFFFFFFU) {
957 continue;
958 }
959 saw_any_frame = true;
960
961 if (!VerifyFrameCrc(candidate)) {
962 continue;
963 }
964
965 if (candidate.rx_common.reply_mode == 0x02) {
966 /* Critical fault frame (UV / clock / supplies). The device is answering
967 * truthfully that it cannot serve the read, so this is not retried. */
968 NoteCriticalFault(candidate);
969 return tle::unexpected(CommError::BusError);
970 }
971
972 const bool width_ok =
973 (expected == ExpectedReply::Bits22) ? (candidate.rx_common.reply_mode == 0x01)
974 : (candidate.rx_common.reply_mode == 0x00);
975 if (!width_ok) {
976 continue; /* Another register's reply landed in this slot. */
977 }
978
979 return (candidate.rx_common.reply_mode == 0x01)
980 ? static_cast<uint32_t>(candidate.rx_22bit.data)
981 : static_cast<uint32_t>(candidate.rx_16bit.data);
982 }
983
984 const uint32_t rx_words[2] = {last_flush, last_reply};
985
986 /* ICVID-only: recover a valid device ID from a misaligned 32-bit MISO word
987 * on long soft-CS / Mode1 buses (data may sit in bits[23:8], or appear one
988 * bit early as 0x82xx/0x80xx/0xC0xx). Restricted to ICVID so lane noise
989 * cannot false-match 0xC1xx inside other registers. */
990 if (address == CentralReg::ICVID) {
991 auto extract_icvid_candidate = [](uint32_t word) noexcept -> uint16_t {
992 if (word == 0U || word == 0xFFFFFFFFU) {
993 return 0;
994 }
995 SPIFrame f{};
996 f.word = word;
997 const uint16_t cands[] = {
998 static_cast<uint16_t>(f.rx_16bit.data),
999 static_cast<uint16_t>((word >> 8) & 0xFFFFU),
1000 static_cast<uint16_t>((word >> 16) & 0xFFFFU),
1001 static_cast<uint16_t>(word & 0xFFFFU),
1002 };
1003 for (uint16_t c : cands) {
1004 if (DeviceID::IsValidDevice(c)) {
1005 return c;
1006 }
1007 /* One-bit-early Mode1 phantoms of a real 0xC1xx ID. */
1008 if ((c & 0xFF00U) == 0x8200U || (c & 0xFF00U) == 0x8000U ||
1009 (c & 0xFF00U) == 0xC000U) {
1010 const uint16_t unshift =
1011 static_cast<uint16_t>((static_cast<uint32_t>(c) << 1) & 0xFFFFU);
1012 if (DeviceID::IsValidDevice(unshift)) {
1013 return unshift;
1014 }
1015 }
1016 }
1017 return 0;
1018 };
1019 for (uint32_t word : {rx_words[1], rx_words[0]}) {
1020 const uint16_t id = extract_icvid_candidate(word);
1021 if (id != 0U) {
1022 return static_cast<uint32_t>(id);
1023 }
1024 }
1025 }
1026
1027 /* Every attempt produced a frame that could not be trusted. Reporting the
1028 * failure is the whole point: the caller can retry, hold its last known
1029 * value, or degrade — all of which beat decoding a corrupt word as state.
1030 *
1031 * `verify_crc` no longer gates this. It used to let callers opt out of
1032 * validation, and every diagnostic path did (`ReadRegister(addr, false)`),
1033 * which is why corrupt and stale frames reached the decoders in the first
1034 * place. The parameter is retained for API compatibility. */
1035 (void)verify_crc;
1036 return tle::unexpected(saw_any_frame ? CommError::CRCError
1037 : CommError::NoReply);
1038}

1040template <typename Derived>
1041inline CommResult<void> SpiInterface<Derived>::ReadMulti(
1042 std::span<const uint16_t> addresses, std::span<uint32_t> values,
1043 uint16_t& valid_mask) noexcept {
1044 valid_mask = 0U;
1045 const size_t n = addresses.size();
1046 if (n == 0U || n > kMaxPipelinedReads || values.size() != n) {
1047 return tle::unexpected(CommError::InvalidParameter);
1048 }
1049
1050 SPIFrame dummy_frame = SPIFrame::MakeRead(0);
1051 dummy_frame.tx_fields.crc = CalculateFrameCrc(dummy_frame);
1052
1053 /* [flush dummy][cmd 0..n-1][trailing dummy]: the reply to command i is
1054 * clocked out during frame i+2. See the ReadMulti doc comment. */
1055 uint32_t tx_words[kMaxPipelinedReads + 2] = {};
1056 uint32_t rx_words[kMaxPipelinedReads + 2] = {};
1057 tx_words[0] = dummy_frame.word;
1058 for (size_t i = 0; i < n; ++i) {
1059 SPIFrame cmd = SPIFrame::MakeRead(addresses[i]);
1060 cmd.tx_fields.crc = CalculateFrameCrc(cmd);
1061 tx_words[i + 1] = cmd.word;
1062 values[i] = 0U;
1063 }
1064 tx_words[n + 1] = dummy_frame.word;
1065
1066 const size_t frames = n + 2U;
1067 auto multi = static_cast<Derived*>(this)->TransferMulti(
1068 std::span<const uint32_t>{tx_words, frames},
1069 std::span<uint32_t>{rx_words, frames});
1070 if (!multi) {
1071 return tle::unexpected(multi.error());
1072 }
1073
1074 for (size_t i = 0; i < n; ++i) {
1075 SPIFrame candidate{};
1076 candidate.word = rx_words[i + 2U];
1077 if (candidate.word == 0U || candidate.word == 0xFFFFFFFFU) {
1078 continue; /* Undriven MISO. */
1079 }
1080 if (!VerifyFrameCrc(candidate)) {
1081 continue;
1082 }
1083 if (candidate.rx_common.reply_mode == 0x02) {
1084 /* Critical fault (UV / clock / supplies). The device is telling us it
1085 * cannot serve any of these reads, so abandon the whole burst. */
1086 NoteCriticalFault(candidate);
1087 return tle::unexpected(CommError::BusError);
1088 }
1089 const ExpectedReply expected = ExpectedReplyFor(addresses[i]);
1090 const bool width_ok = (expected == ExpectedReply::Bits22)
1091 ? (candidate.rx_common.reply_mode == 0x01)
1092 : (candidate.rx_common.reply_mode == 0x00);
1093 if (!width_ok) {
1094 continue; /* Another register's reply landed in this slot. */
1095 }
1096 values[i] = (candidate.rx_common.reply_mode == 0x01)
1097 ? static_cast<uint32_t>(candidate.rx_22bit.data)
1098 : static_cast<uint32_t>(candidate.rx_16bit.data);
1099 valid_mask = static_cast<uint16_t>(valid_mask | (1U << i));
1100 }
1101 return {};
1102}

1104template <typename Derived>
1105inline CommResult<void> SpiInterface<Derived>::Write(uint16_t address, uint16_t value,
1106 bool verify_crc) noexcept {
1107 // Create write frame
1108 SPIFrame tx_frame = SPIFrame::MakeWrite(address, value);
1109
1110 // Calculate and set CRC
1111 tx_frame.tx_fields.crc = CalculateFrameCrc(tx_frame);
1112
1113 /* Leading flush dummy then a trailing dummy to clock the write response —
1114 * same self-synchronising pattern as Read(), so the status bits checked
1115 * below belong to this write and not to whatever ran before it. (See Read()
1116 * for why the dummies must not repeat the accessed address.) */
1117 SPIFrame dummy_frame = SPIFrame::MakeRead(0);
1118 dummy_frame.tx_fields.crc = CalculateFrameCrc(dummy_frame);
1119
1120 const uint32_t tx_words[3] = {dummy_frame.word, tx_frame.word,
1121 dummy_frame.word};
1122 uint32_t rx_words[3] = {};
1123 auto multi = static_cast<Derived*>(this)->TransferMulti(
1124 std::span<const uint32_t>{tx_words}, std::span<uint32_t>{rx_words});
1125 if (!multi) {
1126 return tle::unexpected(multi.error());
1127 }
1128
1129 // Response to the write frame lands in the last slot.
1130 SPIFrame rx_frame{};
1131 rx_frame.word = rx_words[2];
1132
1133 // Verify CRC if requested
1134 if (verify_crc && !VerifyFrameCrc(rx_frame)) {
1135 return tle::unexpected(CommError::CRCError);
1136 }
1137
1138 // Check for errors in status field (for 16-bit reply frames)
1139 if (rx_frame.rx_common.reply_mode == 0x00) {
1140 // Check status field for errors
1141 if (rx_frame.rx_16bit.status != 0x00) {
1142 // Status indicates an error
1143 return tle::unexpected(CommError::TransferError);
1144 }
1145 } else if (rx_frame.rx_common.reply_mode == 0x02) {
1146 // Critical fault frame — keep the cause byte for the caller.
1147 NoteCriticalFault(rx_frame);
1148 return tle::unexpected(CommError::BusError);
1149 }
1150
1151 return {};
1152}
