181 auto Transfer32(uint32_t tx_data) noexcept -> CommResult<uint32_t> {
182 if (!initialized_) {
183 ESP_LOGE(TAG, "CommInterface not initialized");
184 return tle::unexpected(CommError::HardwareNotReady);
185 }
186 uint32_t tx_data_swapped = byte_swap_32(tx_data);
187 uint32_t rx_data_raw = 0;
188 spi_transaction_t trans = {};
189 trans.length = 32;
190 trans.tx_buffer = &tx_data_swapped;
191 trans.rx_buffer = &rx_data_raw;
192
193#if ESP32_TLE_COMM_ENABLE_DETAILED_SPI_LOGGING
194 uint8_t* tx_bytes_orig = reinterpret_cast<uint8_t*>(&tx_data);
195 uint8_t* tx_bytes_swapped = reinterpret_cast<uint8_t*>(&tx_data_swapped);
196 uint8_t crc_tx = tx_bytes_orig[3];
197 uint8_t addr_byte = tx_bytes_orig[2];
198 uint8_t data_high = tx_bytes_orig[1];
199 uint8_t data_low = tx_bytes_orig[0];
200 uint8_t address_upper = (addr_byte >> 1) & 0x7F;
201 bool is_write = (addr_byte & 0x01) != 0;
202 uint16_t data_16bit = (static_cast<uint16_t>(data_high) << 8) | data_low;
203 ESP_LOGI(TAG, "SPI TX Frame: 0x%08X | Swapped: 0x%08X", tx_data, tx_data_swapped);
204 SPIFrame tx_frame_for_crc;
205 tx_frame_for_crc.word = tx_data;
206 uint8_t tx_crc_received = tx_frame_for_crc.tx_fields.crc;
207 tx_frame_for_crc.tx_fields.crc = 0;
208 uint8_t tx_crc_calculated = CalculateFrameCrc(tx_frame_for_crc);
209 bool tx_crc_match = (tx_crc_received == tx_crc_calculated);
210 ESP_LOGI(TAG, " CRC=0x%02X calc=0x%02X %s | Addr=0x%02X %s | Data=0x%04X",
211 tx_crc_received, tx_crc_calculated, tx_crc_match ? "OK" : "MISMATCH",
212 address_upper, is_write ? "W" : "R", data_16bit);
213#endif
214 esp_err_t ret = spi_device_transmit(spi_device_, &trans);
215 if (ret != ESP_OK) {
216 ESP_LOGE(TAG, "SPI transfer failed: %s", esp_err_to_name(ret));
217 return tle::unexpected(CommError::TransferError);
218 }
219 uint32_t rx_data = byte_swap_32(rx_data_raw);
220#if ESP32_TLE_COMM_ENABLE_DETAILED_SPI_LOGGING
221 SPIFrame rx_frame;
222 rx_frame.word = rx_data;
223 uint8_t reply_mode = rx_frame.rx_common.reply_mode;
224 const char* reply_mode_str;
225 switch (reply_mode) {
226 case 0x00: reply_mode_str = "16-bit Reply"; break;
227 case 0x01: reply_mode_str = "22-bit Reply"; break;
228 case 0x02: reply_mode_str = "Critical Fault"; break;
229 default: reply_mode_str = "Unknown"; break;
230 }
231 ESP_LOGI(TAG, "SPI RX Frame: 0x%08X (raw: 0x%08X) | %s", rx_data, rx_data_raw, reply_mode_str);
232 if (reply_mode != 0x02) {
233 bool rx_crc_valid = VerifyFrameCrc(rx_frame);
234 ESP_LOGI(TAG, " CRC %s", rx_crc_valid ? "OK" : "MISMATCH");
235 }
236 if (reply_mode == 0x00) {
237 ESP_LOGI(TAG, " Status=0x%02X RW=%d Data=0x%04X",
238 rx_frame.rx_16bit.status, rx_frame.rx_16bit.rw_echo, rx_frame.rx_16bit.data);
239 } else if (reply_mode == 0x01) {
240 ESP_LOGI(TAG, " 22-bit Data=0x%06X", rx_frame.rx_22bit.data);
241 } else if (reply_mode == 0x02) {
242 auto fault_flags = CriticalFaultFlags::Extract(rx_frame);
243 ESP_LOGI(TAG, " Faults: 1V5=%d 2V5=%d BG=%d CLK_SLOW=%d CLK_FAST=%d",
244 fault_flags.supply_1v5_ok, fault_flags.supply_2v5_ok,
245 fault_flags.adc_bandgap_ok, fault_flags.clk_too_slow, fault_flags.clk_too_fast);
246 }
247#endif
248 return rx_data;
249 }

349 auto GpioSet(CtrlPin pin, GpioSignal signal) noexcept -> CommResult<void> {
350 if (!IsReady()) {
351 last_error_ = CommError::HardwareNotReady;
352 return tle::unexpected(CommError::HardwareNotReady);
353 }
354 int gpio_pin = -1;
355 switch (pin) {
356 case CtrlPin::RESN: gpio_pin = config_.resn_pin; break;
357 case CtrlPin::EN: gpio_pin = config_.en_pin; break;
358 case CtrlPin::FAULTN:
359 last_error_ = CommError::InvalidParameter;
360 return tle::unexpected(CommError::InvalidParameter);
361 }
362 if (gpio_pin < 0) {
363 ESP_LOGE(TAG, "GPIO pin not configured for %s", pin == CtrlPin::RESN ? "RESN" : "EN");
364 last_error_ = CommError::InvalidParameter;
365 return tle::unexpected(CommError::InvalidParameter);
366 }
367 // Map GpioSignal to physical level based on pin active-level:
368 // RESN: active-low → ACTIVE=0, INACTIVE=1
369 // EN: active-high → ACTIVE=1, INACTIVE=0
370 int gpio_level;
371 switch (pin) {
372 case CtrlPin::RESN:
373 gpio_level = (signal == GpioSignal::ACTIVE) ? 0 : 1; // Active-low
374 break;
375 case CtrlPin::EN:
376 gpio_level = (signal == GpioSignal::ACTIVE) ? 1 : 0; // Active-high
377 break;
378 default:
379 return tle::unexpected(CommError::InvalidParameter);
380 }
381 esp_err_t ret = gpio_set_level(static_cast<gpio_num_t>(gpio_pin), gpio_level);
382 if (ret != ESP_OK) {
383 ESP_LOGE(TAG, "Failed to set GPIO%d level: %s", gpio_pin, esp_err_to_name(ret));
384 last_error_ = CommError::BusError;
385 return tle::unexpected(CommError::BusError);
386 }
387 ESP_LOGD(TAG, "Set %s pin (GPIO%d) to %s",
388 pin == CtrlPin::RESN ? "RESN" : "EN", gpio_pin,
389 signal == GpioSignal::ACTIVE ? "ACTIVE" : "INACTIVE");
390 return {};
391 }

419 ESP_LOGD(TAG, "Read FAULTN pin (GPIO%d): %s", gpio_pin,
420 signal == GpioSignal::ACTIVE ? "FAULT" : "NO FAULT");

462 auto initializeGPIO() noexcept -> CommResult<void> {
463 auto configure_output_pin = [this](int16_t pin, const char* name, int initial_level) -> CommResult<void> {
464 if (pin < 0) return {};
465 gpio_config_t cfg = {
466 .pin_bit_mask = (1ULL << pin),
467 .mode = GPIO_MODE_OUTPUT,
468 .pull_up_en = GPIO_PULLUP_DISABLE,
469 .pull_down_en = GPIO_PULLDOWN_DISABLE,
470 .intr_type = GPIO_INTR_DISABLE
471 };
472 if (gpio_config(&cfg) != ESP_OK) {
473 ESP_LOGE(TAG, "Failed to configure %s pin (GPIO%d)", name, pin);
474 return tle::unexpected(CommError::HardwareNotReady);
475 }
476 gpio_set_level(static_cast<gpio_num_t>(pin), initial_level);
477 ESP_LOGI(TAG, "%s pin (GPIO%d) initialized, level=%d", name, pin, initial_level);
478 return {};
479 };
480 if (auto r = configure_output_pin(config_.resn_pin, "RESN", 0); !r) return r;
481 if (auto r = configure_output_pin(config_.en_pin, "EN", 0); !r) return r;
482 if (auto r = configure_output_pin(config_.drv0_pin, "DRV0", 0); !r) return r;
483 if (auto r = configure_output_pin(config_.drv1_pin, "DRV1", 0); !r) return r;
484 if (config_.faultn_pin >= 0) {
485 gpio_config_t cfg = {
486 .pin_bit_mask = (1ULL << config_.faultn_pin),
487 .mode = GPIO_MODE_INPUT,
488 .pull_up_en = GPIO_PULLUP_ENABLE,
489 .pull_down_en = GPIO_PULLDOWN_DISABLE,
490 .intr_type = GPIO_INTR_DISABLE
491 };
492 if (gpio_config(&cfg) != ESP_OK) {
493 ESP_LOGE(TAG, "Failed to configure FAULTN pin (GPIO%d)", config_.faultn_pin);
494 return tle::unexpected(CommError::HardwareNotReady);
495 }
496 ESP_LOGI(TAG, "FAULTN pin (GPIO%d) initialized as input", config_.faultn_pin);
497 }
498 return {};
499 }
