185 auto Transfer32(uint32_t tx_data) noexcept -> CommResult<uint32_t> {
186 if (!initialized_) {
187 ESP_LOGE(TAG, "CommInterface not initialized");
188 return tle::unexpected(CommError::HardwareNotReady);
189 }
190 uint32_t tx_data_swapped = byte_swap_32(tx_data);
191 uint32_t rx_data_raw = 0;
192 spi_transaction_t trans = {};
193 trans.length = 32;
194 trans.tx_buffer = &tx_data_swapped;
195 trans.rx_buffer = &rx_data_raw;
196
197#if ESP32_TLE_COMM_ENABLE_DETAILED_SPI_LOGGING
198 uint8_t* tx_bytes_orig = reinterpret_cast<uint8_t*>(&tx_data);
199 uint8_t* tx_bytes_swapped = reinterpret_cast<uint8_t*>(&tx_data_swapped);
200 uint8_t crc_tx = tx_bytes_orig[3];
201 uint8_t addr_byte = tx_bytes_orig[2];
202 uint8_t data_high = tx_bytes_orig[1];
203 uint8_t data_low = tx_bytes_orig[0];
204 uint8_t address_upper = (addr_byte >> 1) & 0x7F;
205 bool is_write = (addr_byte & 0x01) != 0;
206 uint16_t data_16bit = (static_cast<uint16_t>(data_high) << 8) | data_low;
207 ESP_LOGI(TAG, "SPI TX Frame: 0x%08X | Swapped: 0x%08X", tx_data, tx_data_swapped);
208 SPIFrame tx_frame_for_crc;
209 tx_frame_for_crc.word = tx_data;
210 uint8_t tx_crc_received = tx_frame_for_crc.tx_fields.crc;
211 tx_frame_for_crc.tx_fields.crc = 0;
212 uint8_t tx_crc_calculated = CalculateFrameCrc(tx_frame_for_crc);
213 bool tx_crc_match = (tx_crc_received == tx_crc_calculated);
214 ESP_LOGI(TAG, " CRC=0x%02X calc=0x%02X %s | Addr=0x%02X %s | Data=0x%04X",
215 tx_crc_received, tx_crc_calculated, tx_crc_match ? "OK" : "MISMATCH",
216 address_upper, is_write ? "W" : "R", data_16bit);
217#endif
218 esp_err_t ret = spi_device_transmit(spi_device_, &trans);
219 if (ret != ESP_OK) {
220 ESP_LOGE(TAG, "SPI transfer failed: %s", esp_err_to_name(ret));
221 return tle::unexpected(CommError::TransferError);
222 }
223 uint32_t rx_data = byte_swap_32(rx_data_raw);
224#if ESP32_TLE_COMM_ENABLE_DETAILED_SPI_LOGGING
225 SPIFrame rx_frame;
226 rx_frame.word = rx_data;
227 uint8_t reply_mode = rx_frame.rx_common.reply_mode;
228 const char* reply_mode_str;
229 switch (reply_mode) {
230 case 0x00: reply_mode_str = "16-bit Reply"; break;
231 case 0x01: reply_mode_str = "22-bit Reply"; break;
232 case 0x02: reply_mode_str = "Critical Fault"; break;
233 default: reply_mode_str = "Unknown"; break;
234 }
235 ESP_LOGI(TAG, "SPI RX Frame: 0x%08X (raw: 0x%08X) | %s", rx_data, rx_data_raw, reply_mode_str);
236 if (reply_mode != 0x02) {
237 bool rx_crc_valid = VerifyFrameCrc(rx_frame);
238 ESP_LOGI(TAG, " CRC %s", rx_crc_valid ? "OK" : "MISMATCH");
239 }
240 if (reply_mode == 0x00) {
241 ESP_LOGI(TAG, " Status=0x%02X RW=%d Data=0x%04X",
242 rx_frame.rx_16bit.status, rx_frame.rx_16bit.rw_echo, rx_frame.rx_16bit.data);
243 } else if (reply_mode == 0x01) {
244 ESP_LOGI(TAG, " 22-bit Data=0x%06X", rx_frame.rx_22bit.data);
245 } else if (reply_mode == 0x02) {
246 auto fault_flags = CriticalFaultFlags::Extract(rx_frame);
247 ESP_LOGI(TAG, " Faults: 1V5=%d 2V5=%d BG=%d CLK_SLOW=%d CLK_FAST=%d",
248 fault_flags.supply_1v5_ok, fault_flags.supply_2v5_ok,
249 fault_flags.adc_bandgap_ok, fault_flags.clk_too_slow, fault_flags.clk_too_fast);
250 }
251#endif
252 return rx_data;
253 }

353 auto GpioSet(CtrlPin pin, GpioSignal signal) noexcept -> CommResult<void> {
354 if (!IsReady()) {
355 last_error_ = CommError::HardwareNotReady;
356 return tle::unexpected(CommError::HardwareNotReady);
357 }
358 int gpio_pin = -1;
359 switch (pin) {
360 case CtrlPin::RESN: gpio_pin = config_.resn_pin; break;
361 case CtrlPin::EN: gpio_pin = config_.en_pin; break;
362 case CtrlPin::FAULTN:
363 last_error_ = CommError::InvalidParameter;
364 return tle::unexpected(CommError::InvalidParameter);
365 }
366 if (gpio_pin < 0) {
367 ESP_LOGE(TAG, "GPIO pin not configured for %s", pin == CtrlPin::RESN ? "RESN" : "EN");
368 last_error_ = CommError::InvalidParameter;
369 return tle::unexpected(CommError::InvalidParameter);
370 }
371 // Map GpioSignal to physical level based on pin active-level:
372 // RESN: active-low → ACTIVE=0, INACTIVE=1
373 // EN: active-high → ACTIVE=1, INACTIVE=0
374 int gpio_level;
375 switch (pin) {
376 case CtrlPin::RESN:
377 gpio_level = (signal == GpioSignal::ACTIVE) ? 0 : 1; // Active-low
378 break;
379 case CtrlPin::EN:
380 gpio_level = (signal == GpioSignal::ACTIVE) ? 1 : 0; // Active-high
381 break;
382 default:
383 return tle::unexpected(CommError::InvalidParameter);
384 }
385 esp_err_t ret = gpio_set_level(static_cast<gpio_num_t>(gpio_pin), gpio_level);
386 if (ret != ESP_OK) {
387 ESP_LOGE(TAG, "Failed to set GPIO%d level: %s", gpio_pin, esp_err_to_name(ret));
388 last_error_ = CommError::BusError;
389 return tle::unexpected(CommError::BusError);
390 }
391 ESP_LOGD(TAG, "Set %s pin (GPIO%d) to %s",
392 pin == CtrlPin::RESN ? "RESN" : "EN", gpio_pin,
393 signal == GpioSignal::ACTIVE ? "ACTIVE" : "INACTIVE");
394 return {};
395 }

420 int gpio_level = gpio_get_level(static_cast<gpio_num_t>(gpio_pin));
421 // FAULTN is active-low: physical 0 = fault active
422 GpioSignal signal = (gpio_level == 0) ? GpioSignal::ACTIVE : GpioSignal::INACTIVE;
423 ESP_LOGD(TAG, "Read FAULTN pin (GPIO%d): %s", gpio_pin,
424 signal == GpioSignal::ACTIVE ? "FAULT" : "NO FAULT");

466 auto initializeGPIO() noexcept -> CommResult<void> {
467 auto configure_output_pin = [this](int16_t pin, const char* name, int initial_level) -> CommResult<void> {
468 if (pin < 0) return {};
469 gpio_config_t cfg = {
470 .pin_bit_mask = (1ULL << pin),
471 .mode = GPIO_MODE_OUTPUT,
472 .pull_up_en = GPIO_PULLUP_DISABLE,
473 .pull_down_en = GPIO_PULLDOWN_DISABLE,
474 .intr_type = GPIO_INTR_DISABLE
475 };
476 if (gpio_config(&cfg) != ESP_OK) {
477 ESP_LOGE(TAG, "Failed to configure %s pin (GPIO%d)", name, pin);
478 return tle::unexpected(CommError::HardwareNotReady);
479 }
480 gpio_set_level(static_cast<gpio_num_t>(pin), initial_level);
481 ESP_LOGI(TAG, "%s pin (GPIO%d) initialized, level=%d", name, pin, initial_level);
482 return {};
483 };
484 if (auto r = configure_output_pin(config_.resn_pin, "RESN", 0); !r) return r;
485 if (auto r = configure_output_pin(config_.en_pin, "EN", 0); !r) return r;
486 if (auto r = configure_output_pin(config_.drv0_pin, "DRV0", 0); !r) return r;
487 if (auto r = configure_output_pin(config_.drv1_pin, "DRV1", 0); !r) return r;
488 if (config_.faultn_pin >= 0) {
489 gpio_config_t cfg = {
490 .pin_bit_mask = (1ULL << config_.faultn_pin),
491 .mode = GPIO_MODE_INPUT,
492 .pull_up_en = GPIO_PULLUP_ENABLE,
493 .pull_down_en = GPIO_PULLDOWN_DISABLE,
494 .intr_type = GPIO_INTR_DISABLE
495 };
496 if (gpio_config(&cfg) != ESP_OK) {
497 ESP_LOGE(TAG, "Failed to configure FAULTN pin (GPIO%d)", config_.faultn_pin);
498 return tle::unexpected(CommError::HardwareNotReady);
499 }
500 ESP_LOGI(TAG, "FAULTN pin (GPIO%d) initialized as input", config_.faultn_pin);
501 }
502 return {};
503 }
