610[[nodiscard]] constexpr uint16_t CalculateTarget(uint16_t current_ma,
611 bool parallel_mode = false) noexcept {
612 uint32_t max_current = parallel_mode ? 4000 : 2000;
613 // Integer math: adding (max_current/2) before division is equivalent to (val / max_current + 0.5)
614 // for rounding up (uint32_t(current_ma) * 32767UL + max_current / 2) / max_current <=>
615 // (current_ma * 32767 / max_current + 0.5)
616 uint32_t target = (static_cast<uint32_t>(current_ma) * 32767UL + (max_current / 2)) / max_current;
617 // Saturate at MAX_TARGET
618 target = std::min(target, static_cast<uint32_t>(MAX_TARGET));
619 return static_cast<uint16_t>(target);
620}
621
630[[nodiscard]] constexpr uint16_t CalculateCurrent(uint16_t target,
631 bool parallel_mode = false) noexcept {
632 uint32_t max_current = parallel_mode ? 4000 : 2000;
633 // Use rounding: add half of 32767 before division to minimize quantization error
634 uint32_t current = (static_cast<uint32_t>(target & TARGET_MASK) * max_current + 16383) / 32767UL;
635 return static_cast<uint16_t>(current);
636}
637
638// Datasheet §5.3.3.3: I_set = 2 A * TARGET / (2^15-1); 0x6000 saturates at 1.5 A.
639// A 9-bit TARGET (0x1FF) decodes as 31 mA — the HIL "clamped to 31 mA" signature
640// when SETPOINT readback is actually FB_IMIN (10-bit signed, max +511) or a
641// truncated field rather than the 15-bit TARGET. CH_CTRL 0x8020 (Mission+CH5)
642// decodes as 2 mA — the leftover from a MakeRead(0) dummy.
643static_assert(CalculateTarget(115) == 1884, "115 mA → TARGET 0x075C");
644static_assert(CalculateCurrent(1884) == 115, "TARGET 0x075C → 115 mA");
645static_assert(CalculateCurrent(0x01FF) == 31, "9-bit max TARGET decodes as 31 mA");
646static_assert(CalculateCurrent(0x0020) == 2, "CH_CTRL EN_CH5 leftover decodes as 2 mA");

687//
688// Datasheet-spec PWM frequency range
689// ----------------------------------
690// Per datasheet (Electrical Characteristics, parameter "Target PWM frequency
691// fPWM"): the PWM frequency controller is specified to operate between
692// 110 Hz and 4000 Hz. Setting a value outside this range will not be
693// rejected by the chip — it simply isn't guaranteed to work.
694//
695// With the LOW_FREQ_RANGE_EN bit set, the period is multiplied by 8, so
696// the supported frequency range is 8× lower (≈13.75 Hz – 500 Hz).
697//
698// Periods are derived from the formula T_pwm = 1 / f_pwm.
699//
700// Note: the chip family cannot drive PWM above 4 kHz, so reaching the
701// ultrasonic range (>20 kHz) is not possible with the TLE92466ED.
702//
703constexpr float kSpecMinFrequency_Hz = 110.0F;
704constexpr float kSpecMaxFrequency_Hz = 4000.0F;
705constexpr float kSpecMinPeriod_us = 1.0e6F / kSpecMaxFrequency_Hz;
706constexpr float kSpecMaxPeriod_us = 1.0e6F / kSpecMinFrequency_Hz;
707
708constexpr float kSpecLowRangeMinFreq_Hz = kSpecMinFrequency_Hz / 8.0F;
709constexpr float kSpecLowRangeMaxFreq_Hz = kSpecMaxFrequency_Hz / 8.0F;
710constexpr float kSpecLowRangeMinPeriod_us =
711 1.0e6F / kSpecLowRangeMaxFreq_Hz;
712constexpr float kSpecLowRangeMaxPeriod_us =
713 1.0e6F / kSpecLowRangeMinFreq_Hz;
714
716constexpr float kSpecCombinedMinPeriod_us = kSpecMinPeriod_us;
718constexpr float kSpecCombinedMaxPeriod_us = kSpecLowRangeMaxPeriod_us;

733struct PeriodConfig {
734 uint8_t mantissa;
735 uint8_t exponent;
736 bool low_freq_range;
737
742 [[nodiscard]] constexpr float CalculatePeriodUs() const noexcept {
743 float base_period =
744 static_cast<float>(mantissa) * static_cast<float>(1ULL << exponent) * F_SYS_PERIOD_US;
745 return low_freq_range ? (base_period * 8.0F) : base_period;
746 }
747};
748
754[[nodiscard]] inline PeriodConfig CalculateFromPeriodUs(float period_us) noexcept {
755 PeriodConfig config{};
756
757 // Try standard range first
758 for (uint8_t exp = 0; exp <= 7; ++exp) {
759 float divisor = static_cast<float>(1ULL << exp) * F_SYS_PERIOD_US;
760 float mantissa_f = period_us / divisor;
761
762 if (mantissa_f <= 255.0F && mantissa_f >= 1.0F) {
763 config.mantissa = static_cast<uint8_t>(std::lround(mantissa_f)); // Round
764 config.exponent = exp;
765 config.low_freq_range = false;
766 return config;
767 }
768 }
769
770 // Try low frequency range (8x multiplier)
771 for (uint8_t exp = 0; exp <= 7; ++exp) {
772 float divisor = static_cast<float>(1ULL << exp) * F_SYS_PERIOD_US * 8.0F;
773 float mantissa_f = period_us / divisor;
774
775 if (mantissa_f <= 255.0F && mantissa_f >= 1.0F) {
776 config.mantissa = static_cast<uint8_t>(std::lround(mantissa_f)); // Round
777 config.exponent = exp;
778 config.low_freq_range = true;
779 return config;
780 }
781 }
782
783 // Out of range - return invalid (mantissa = 0)
784 config.mantissa = 0;
785 config.exponent = 0;
786 config.low_freq_range = false;
787 return config;
788}

795[[nodiscard]] constexpr uint16_t BuildRegisterValue(const PeriodConfig& config) noexcept {
796 return config.mantissa | ((config.exponent & 0x07) << EXP_SHIFT) |
797 (config.low_freq_range ? LOW_FREQ_BIT : 0);
798}

1288 const float mant_f = t_ref_clk_us / divisor;

1290 if (mant_f > 1023.0F) continue; // out of 10-bit range

1298 }
1299 }

1342 static_cast<float>(max_current)) /
1343 32767.0F;
1344 return amplitude;
1345 }
1346

1353 float t_ref_clk_us = DEFAULT_T_REF_CLK_US) const noexcept {

1355 t_ref_clk_us;
1356 }
1357};
1358
1373[[nodiscard]] inline DitherConfig CalculateFromAmplitudeFrequency(
1374 float amplitude_ma, // Current amplitude in milliamperes
1375 float frequency_hz, // Dither frequency in hertz
1376 bool parallel_mode = false,
1377 float t_ref_clk_us = DEFAULT_T_REF_CLK_US) noexcept {
1378
1379 DitherConfig config{};
1380
1381 // Calculate desired period from frequency
1382 float period_us = 1'000'000.0F / frequency_hz;
1383
1384 // Calculate dither period from formula: T_dither = [4×STEPS + 2×FLAT] × t_ref_clk
1385 // Use reasonable defaults: num_steps = 16, flat_steps = 2
1386 constexpr uint8_t default_steps = 16;
1387 constexpr uint8_t default_flat = 2;
1388
1389 float calculated_period = (4.0F * default_steps + 2.0F * default_flat) * t_ref_clk_us;
1390
1391 // If calculated period doesn't match desired, adjust steps
1392 if (calculated_period < period_us * 0.9F || calculated_period > period_us * 1.1F) {
1393 // Adjust num_steps to get closer to desired period
1394 float target_steps = (period_us / t_ref_clk_us - 2.0F * default_flat) / 4.0F;
1395 // Clamp to valid range before casting to uint8_t
1396 target_steps = std::max(1.0F, target_steps);
1397 target_steps = std::min(255.0F, target_steps);
1398 config.num_steps = static_cast<uint8_t>(std::lround(target_steps));
1399 } else {
1400 config.num_steps = default_steps;
1401 }
1402
1403 config.flat_steps = default_flat;
1404
1405 // Calculate step_size from amplitude formula: I_dither = STEPS × STEP_SIZE × 2A / 32767
1406 uint32_t max_current = parallel_mode ? 4000 : 2000;
1407 float step_size_f = (amplitude_ma * 32767.0F) /
1408 (static_cast<float>(config.num_steps) * static_cast<float>(max_current));
1409
1410 config.step_size = static_cast<uint16_t>(std::lround(step_size_f));
1411 config.step_size = std::min(config.step_size, static_cast<uint16_t>(DITHER_CTRL::STEP_SIZE_MASK));
1412
1413 return config;
1414}
